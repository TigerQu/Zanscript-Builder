===== 02_mwp_pd.zs =====
# Step 2: Plate Arena & Dual-Zone Asset Builder (Petri Dish)
# Config profile: Default MWP
# Place an EMPTY Petri Dish plate into the MWP stand before running.
# Generates "aPD.bmp" (full wells) and "dzPD.bmp" (50/50 outer/inner zones) in Assets.

DEFINE SHOWTIME 2
DEFINE DRAWDELAY 0.4

# Plate geometry in mm (1 rows x 1 columns)
DEFINE X_OFFSET 67
DEFINE Y_OFFSET 43
DEFINE COL_STEP 19.3
DEFINE ROW_STEP 19.3
DEFINE RADIUS 44
DEFINE RADIUS_DZ 31.8

DEFINE ARENA_ZONE_MODE 200
DEFINE ROW_COUNTER 101
DEFINE ARENA_COUNTER 102
DEFINE COLUMN_COUNTER 103
DEFINE Y_POSITION 104
DEFINE X_POSITION 105

ACTION MAIN

    INVOKE(BUILD_ARENAS,1)
    WAIT(SHOWTIME)

    INVOKE(BUILD_DUAL_ZONES,1)
    WAIT(SHOWTIME)

COMPLETE

ACTION BUILD_ARENAS

    ResetDrawing()
    ShapeType(DISC,RADIUS)
    @ARENA_ZONE_MODE = 0
    INVOKE(DRAW_GRID,1)
    SaveDrawing("aPD")

COMPLETE

ACTION BUILD_DUAL_ZONES

    ResetDrawing()
    ShapeType(DISC,RADIUS)
    @ARENA_ZONE_MODE = 1
    INVOKE(DRAW_GRID,1)

    ShapeType(DISC,RADIUS_DZ)
    @ARENA_ZONE_MODE = 2
    INVOKE(DRAW_GRID,1)
    SaveDrawing("dzPD")

COMPLETE

ACTION DRAW_GRID

    @ROW_COUNTER = 0
    @Y_POSITION = Y_OFFSET
    @ARENA_COUNTER = 1

    WHILE @ROW_COUNTER < 1
        @COLUMN_COUNTER = 0
        @X_POSITION = X_OFFSET

        WHILE @COLUMN_COUNTER < 1
            IF @ARENA_ZONE_MODE = 0
                SET(DrawArea,@ARENA_COUNTER)
            ELSE
                SET(DrawArea,@ARENA_ZONE_MODE)
            ENDIF

            ShapeDraw(@X_POSITION,@Y_POSITION)
            SaveDrawing("TEMP")
            WAIT(DRAWDELAY)

            @X_POSITION = @X_POSITION + COL_STEP
            @COLUMN_COUNTER = @COLUMN_COUNTER + 1
            @ARENA_COUNTER = @ARENA_COUNTER + 1
        ENDWHILE

        @Y_POSITION = @Y_POSITION + ROW_STEP
        @ROW_COUNTER = @ROW_COUNTER + 1
    ENDWHILE

COMPLETE


===== 03_mwp_pd.zs =====
# Step 3: Organism Tracking Check (Petri Dish)
# Optional 1-2 minute pre-check with animals loaded in the plate.
# Sweeps TARGET_SIZE (2 to 5) and DETECTOR_THRESHOLD (2 to 10)
# so you can confirm clean target lock on video before running Step 4.

DEFINE TIME_BIN 1
DEFINE SAMPLES_PER_SETTING 2
DEFINE MIN_SIZE 2
DEFINE MAX_SIZE 5
DEFINE MIN_THRESH 2
DEFINE MAX_THRESH 10
DEFINE AUTOREF_TIME 60
DEFINE VIDEO_LENGTH 999999
DEFINE X_LED_DRIVE 30039
DEFINE LED_BRIGHTNESS 4095
# SET(THERMOSTAT,21)  # Thermostat disabled

DEFINE SIZE_VAR 101
DEFINE THRESH_VAR 102

SET(AUTOREF_MODE,MOVEMENT)
SET(AUTOREF_TIMEOUT,AUTOREF_TIME)

ACTION MAIN

    SET(COUNTER1,COUNTER_ZERO)
    INVOKE(LIGHTS_ON,1)
    LOAD(ARENAS,"aPD.bmp")

    LOGFILE(0,"mwp_track_check_pd")
    LOGCREATE("TEXT:RUNTIME|TEXT:UNIT_ID|TEXT:TEMP_C|TEXT:TARGET_SIZE|TEXT:DET_THRESHOLD|TEXT:TIME_BIN")
    LOGAPPEND("TEXT:DIST_A1")
    LOGRUN()

    VIDEO(VIDEO_LENGTH,"mwp_track_check_pd")
    LOAD(OVERLAY_TEXT,"1,1:AUTOREFERENCING       ")
    AUTOREFERENCE()

    INVOKE(SWEEP_SETTINGS,1)

    VIDEOSTOP
    LOAD(OVERLAY_CLEAR,"")
    INVOKE(LIGHTS_OFF,1)

COMPLETE

ACTION SWEEP_SETTINGS

    @SIZE_VAR = MIN_SIZE
    WHILE @SIZE_VAR <= MAX_SIZE
        SET(TARGET_SIZE,@SIZE_VAR)
        TargetMarker(@SIZE_VAR,1,1)
        @THRESH_VAR = MIN_THRESH

        WHILE @THRESH_VAR <= MAX_THRESH
            SET(DETECTOR_THRESHOLD,@THRESH_VAR)
            LOAD(OVERLAY_TEXT,"1,1:SIZE @101  THRESH @102  ")
            INVOKE(SAMPLE_BIN,SAMPLES_PER_SETTING)
            @THRESH_VAR = @THRESH_VAR + 2
        ENDWHILE

        @SIZE_VAR = @SIZE_VAR + 1
    ENDWHILE

COMPLETE

ACTION SAMPLE_BIN

    SET(COUNTER1,COUNTER_INC)
    LOGDATA(DATA_SNAPSHOT,"begin")
    WAIT(TIME_BIN)
    LOGDATA(DATA_SNAPSHOT,"end")
    LOGDATA(DATA_SELECT,"begin")
    LOGDATA(DATA_DELTA,"end")

    LOGCREATE("RUNTIME|APPARATUS_ID|TEMPERATURE1|@101|@102|COUNTER1")
    LOGAPPEND("ARENA_DISTANCES:*")
    LOGRUN()

COMPLETE

ACTION LIGHTS_ON
    SET(X_LED_DRIVE,4,LED_BRIGHTNESS)
    # SET(X_LED_DRIVE,12,LED_BRIGHTNESS)  # Uncomment for underneath LED
COMPLETE

ACTION LIGHTS_OFF
    SET(X_LED_DRIVE,4,0)
    # SET(X_LED_DRIVE,12,0)
COMPLETE


===== 04_mwp_pd.zs =====
# Step 4: Main Assay Script (Petri Dish | Long-Term Cycle)
# Required asset(s) in unit Assets folder: "aPD.bmp"

DEFINE TIME_BIN 30
DEFINE HALF_HOUR_BINS 55  # 2 x 55 x 30s + 300s autoref = ~1 hour
DEFINE LIGHT_HOURS 12
DEFINE DARK_HOURS 12
DEFINE NUM_DAYS 3
DEFINE DAY_VAR 100
DEFINE HOUR_VAR 101
DEFINE PHASE_VAR 102
DEFINE VIDEO_LENGTH 999999999
DEFINE X_LED_DRIVE 30039
DEFINE LED_BRIGHTNESS 4095
DEFINE X_DRAWTRACKS 30011
DEFINE X_TRACKTIME 30016

# SET(THERMOSTAT,21)  # Thermostat disabled
SET(TARGET_SIZE,3)
SET(DETECTOR_THRESHOLD,6)
SET(AUTOREF_MODE,2)
SET(AUTOREF_TIMEOUT,300)
TargetMarker(1.8,1,1)
SET(MSD_MODE,ON)
SET(MSD_THRESHOLD,6)

ACTION MAIN

    SET(COUNTER1,COUNTER_ZERO)
    LOAD(ARENAS,"aPD.bmp")

    LOGFILE(0,"mwp_circadian_12l12d_3d_pd")
    INVOKE(SETUP_HEADERS,1)

    @DAY_VAR = 0
    INVOKE(LIGHTS_ON,1)
    LOAD(OVERLAY_TEXT,"1,1:AUTOREFERENCING       ")
    AUTOREFERENCE()
    SET(X_DRAWTRACKS,1)
    SET(X_TRACKTIME,15)
    VIDEO(VIDEO_LENGTH,"mwp_circadian_12l12d_3d_pd")
    VIDEOSKIP(29)
    INVOKE(RUN_DAY,NUM_DAYS)

    VIDEOSTOP
    LOAD(OVERLAY_CLEAR,"")
    INVOKE(LIGHTS_OFF,1)

COMPLETE

ACTION RUN_DAY

    @DAY_VAR = @DAY_VAR + 1
    @HOUR_VAR = 1
    @PHASE_VAR = 1
    INVOKE(LIGHTS_ON,1)
    INVOKE(RUN_HOUR,LIGHT_HOURS)
    @HOUR_VAR = 1
    @PHASE_VAR = 0
    INVOKE(LIGHTS_OFF,1)
    INVOKE(RUN_HOUR,DARK_HOURS)

COMPLETE

ACTION RUN_HOUR

    IF @PHASE_VAR = 1
        LOAD(OVERLAY_TEXT,"1,1:DAY @100 LIGHT HR @101  ")
    ELSE
        LOAD(OVERLAY_TEXT,"1,1:DAY @100 DARK  HR @101  ")
    ENDIF

    INVOKE(SAMPLE_CIRC,HALF_HOUR_BINS)
    AUTOREFERENCE()
    INVOKE(SAMPLE_CIRC,HALF_HOUR_BINS)
    @HOUR_VAR = @HOUR_VAR + 1

COMPLETE

ACTION SAMPLE_CIRC

    SET(COUNTER1,COUNTER_INC)
    LOGDATA(DATA_SNAPSHOT,"begin")
    WAIT(TIME_BIN)
    LOGDATA(DATA_SNAPSHOT,"end")
    LOGDATA(DATA_SELECT,"begin")
    LOGDATA(DATA_DELTA,"end")

    IF @PHASE_VAR = 1
        LOGCREATE("RUNTIME|APPARATUS_ID|TEMPERATURE1|TEXT:LIGHT|@100|@101|COUNTER1")
    ELSE
        LOGCREATE("RUNTIME|APPARATUS_ID|TEMPERATURE1|TEXT:DARK|@100|@101|COUNTER1")
    ENDIF
    LOGAPPEND("ARENA_DISTANCES:*")
    LOGAPPEND("ARENA_ACTIVITY:*")
    LOGRUN()

COMPLETE

ACTION LIGHTS_ON
    SET(X_LED_DRIVE,4,LED_BRIGHTNESS)
    # SET(X_LED_DRIVE,12,LED_BRIGHTNESS)  # Uncomment for underneath LED
COMPLETE

ACTION LIGHTS_OFF
    SET(X_LED_DRIVE,4,0)
    # SET(X_LED_DRIVE,12,0)
COMPLETE

ACTION SETUP_HEADERS

    LOGCREATE("TEXT:RUNTIME|TEXT:UNIT_ID|TEXT:TEMP_C|TEXT:PHASE|TEXT:DAY|TEXT:HOUR|TEXT:TIME_BIN")
    LOGAPPEND("TEXT:DIST_A1|TEXT:MSD_A1")
    LOGRUN()

COMPLETE

ZANTIKS MWP EXPERIMENT BUNDLE
=============================
Researcher       : Not specified
Study Tag        : (none)
Plate Format     : Petri Dish (1 x 1, 1 arena(s))
Config Profile   : Default
Plate Geometry   : X_OFFSET=67, Y_OFFSET=43, COL_STEP=19.3, ROW_STEP=19.3, RADIUS=44, INNER_ZONE=31.8 mm
Hardware Target  : MWP Z3
Stimulus Protocol: Long-Term Cycle
Estimated Runtime: 72h 05m (7920 CSV rows)

EXECUTION ORDER IN ZANTIKS CONTROL CONSOLE
------------------------------------------
1. 01_cal.zs
   - Run ONCE with the black 96-well calibration plate. Skip if already calibrated.

2. 02_mwp_pd.zs
   - Run ONCE with an EMPTY Petri Dish plate.
   - Generates "aPD.bmp" and "dzPD.bmp" inside the unit's Assets directory.

3. 03_mwp_pd.zs (Optional)
   - Run with organisms loaded to check TARGET_SIZE and DETECTOR_THRESHOLD on Live View.

4. 04_mwp_pd.zs
   - Main assay script. Load your experimental plate, build, and run.
   - Slot name for Settings -> Slots (1-20): "04_mwp_pd" (link once; rebuilding updates it in place).
   - Exported CSV & video files are tagged automatically via LOGFILE(0,"mwp_circadian_12l12d_3d_pd").
   - Download exported .csv data from Control Console -> Data files when complete.

UNIT REMINDERS
--------------
- Do not rename scripts with spaces or parentheses.
- Always click Save before Build in the Zantiks web editor.
- Always shut down via the Control Console webpage before switching off unit power.
