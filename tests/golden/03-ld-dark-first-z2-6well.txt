===== 02_mwp_6w.zs =====
# Step 2: Plate Arena & Dual-Zone Asset Builder (6-Well)
# Config profile: Default MWP
# Place an EMPTY 6-Well plate into the MWP stand before running.
# Generates "a6.bmp" (full wells) and "dz6.bmp" (50/50 outer/inner zones) in Assets.

DEFINE SHOWTIME 2
DEFINE DRAWDELAY 0.3

# Plate geometry in mm (2 rows x 3 columns)
DEFINE X_OFFSET 24.76
DEFINE Y_OFFSET 23.16
DEFINE COL_STEP 39.12
DEFINE ROW_STEP 39.12
DEFINE RADIUS 17.72
DEFINE RADIUS_DZ 12.73

DEFINE ARENA_ZONE_MODE 200
DEFINE ROW_COUNTER 101
DEFINE ARENA_COUNTER 102
DEFINE COLUMN_COUNTER 103
DEFINE Y_POSITION 104
DEFINE X_POSITION 105

ACTION MAIN

    INVOKE(BUILD_ARENAS,1)
    WAIT(SHOWTIME)

    INVOKE(BUILD_DUAL_ZONES,1)
    WAIT(SHOWTIME)

COMPLETE

ACTION BUILD_ARENAS

    ResetDrawing()
    ShapeType(DISC,RADIUS)
    @ARENA_ZONE_MODE = 0
    INVOKE(DRAW_GRID,1)
    SaveDrawing("a6")

COMPLETE

ACTION BUILD_DUAL_ZONES

    ResetDrawing()
    ShapeType(DISC,RADIUS)
    @ARENA_ZONE_MODE = 1
    INVOKE(DRAW_GRID,1)

    ShapeType(DISC,RADIUS_DZ)
    @ARENA_ZONE_MODE = 2
    INVOKE(DRAW_GRID,1)
    SaveDrawing("dz6")

COMPLETE

ACTION DRAW_GRID

    @ROW_COUNTER = 0
    @Y_POSITION = Y_OFFSET
    @ARENA_COUNTER = 1

    WHILE @ROW_COUNTER < 2
        @COLUMN_COUNTER = 0
        @X_POSITION = X_OFFSET

        WHILE @COLUMN_COUNTER < 3
            IF @ARENA_ZONE_MODE = 0
                SET(DrawArea,@ARENA_COUNTER)
            ELSE
                SET(DrawArea,@ARENA_ZONE_MODE)
            ENDIF

            ShapeDraw(@X_POSITION,@Y_POSITION)
            SaveDrawing("TEMP")
            WAIT(DRAWDELAY)

            @X_POSITION = @X_POSITION + COL_STEP
            @COLUMN_COUNTER = @COLUMN_COUNTER + 1
            @ARENA_COUNTER = @ARENA_COUNTER + 1
        ENDWHILE

        @Y_POSITION = @Y_POSITION + ROW_STEP
        @ROW_COUNTER = @ROW_COUNTER + 1
    ENDWHILE

COMPLETE


===== 03_mwp_6w.zs =====
# Step 3: Organism Tracking Check (6-Well)
# Optional 1-2 minute pre-check with animals loaded in the plate.
# Sweeps TARGET_SIZE (4 to 7) and DETECTOR_THRESHOLD (2 to 10)
# so you can confirm clean target lock on video before running Step 4.

DEFINE TIME_BIN 1
DEFINE SAMPLES_PER_SETTING 2
DEFINE MIN_SIZE 4
DEFINE MAX_SIZE 7
DEFINE MIN_THRESH 2
DEFINE MAX_THRESH 10
DEFINE AUTOREF_TIME 60
DEFINE VIDEO_LENGTH 999999
SET(THERMOSTAT,23)

DEFINE SIZE_VAR 101
DEFINE THRESH_VAR 102

SET(AUTOREF_MODE,MOVEMENT)
SET(AUTOREF_TIMEOUT,AUTOREF_TIME)

ACTION MAIN

    SET(COUNTER1,COUNTER_ZERO)
    INVOKE(LIGHTS_OFF,1)
    LOAD(ARENAS,"a6.bmp")

    LOGFILE(0,"mwp_track_check_6w")
    LOGCREATE("TEXT:RUNTIME|TEXT:UNIT_ID|TEXT:TEMP_C|TEXT:TARGET_SIZE|TEXT:DET_THRESHOLD|TEXT:TIME_BIN")
    LOGAPPEND("TEXT:DIST_A1|TEXT:DIST_A2|TEXT:DIST_A3|TEXT:DIST_B1|TEXT:DIST_B2|TEXT:DIST_B3")
    LOGRUN()

    VIDEO(VIDEO_LENGTH,"mwp_track_check_6w")
    LOAD(OVERLAY_TEXT,"1,1:AUTOREFERENCING       ")
    AUTOREFERENCE()

    INVOKE(SWEEP_SETTINGS,1)

    VIDEOSTOP
    LOAD(OVERLAY_CLEAR,"")
    INVOKE(LIGHTS_OFF,1)

COMPLETE

ACTION SWEEP_SETTINGS

    @SIZE_VAR = MIN_SIZE
    WHILE @SIZE_VAR <= MAX_SIZE
        SET(TARGET_SIZE,@SIZE_VAR)
        TargetMarker(@SIZE_VAR,1,1)
        @THRESH_VAR = MIN_THRESH

        WHILE @THRESH_VAR <= MAX_THRESH
            SET(DETECTOR_THRESHOLD,@THRESH_VAR)
            LOAD(OVERLAY_TEXT,"1,1:SIZE @101  THRESH @102  ")
            INVOKE(SAMPLE_BIN,SAMPLES_PER_SETTING)
            @THRESH_VAR = @THRESH_VAR + 2
        ENDWHILE

        @SIZE_VAR = @SIZE_VAR + 1
    ENDWHILE

COMPLETE

ACTION SAMPLE_BIN

    SET(COUNTER1,COUNTER_INC)
    LOGDATA(DATA_SNAPSHOT,"begin")
    WAIT(TIME_BIN)
    LOGDATA(DATA_SNAPSHOT,"end")
    LOGDATA(DATA_SELECT,"begin")
    LOGDATA(DATA_DELTA,"end")

    LOGCREATE("RUNTIME|APPARATUS_ID|TEMPERATURE1|@101|@102|COUNTER1")
    LOGAPPEND("ARENA_DISTANCES:*")
    LOGRUN()

COMPLETE

ACTION LIGHTS_ON
    SET(GPO6,1)
    SET(GPO7,1)
    SET(GPO8,1)
COMPLETE

ACTION LIGHTS_OFF
    LIGHTS(ALL,OFF)
    SET(GPO6,0)
    SET(GPO7,0)
    SET(GPO8,0)
COMPLETE


===== 04_mwp_6w.zs =====
# Step 4: Main Assay Script (6-Well | Light / Dark Cycles)
# Required asset(s) in unit Assets folder: "a6.bmp"

DEFINE ACCLIMATION_SEC 120
DEFINE TIME_BIN 1
DEFINE PHASE_BINS 120
DEFINE NUM_CYCLES 3
DEFINE VIDEO_LENGTH 999999999
DEFINE X_DRAWTRACKS 30011
DEFINE X_TRACKTIME 30016

SET(THERMOSTAT,23)
SET(TARGET_SIZE,5)
SET(DETECTOR_THRESHOLD,3)
SET(AUTOREF_MODE,MOVEMENT)
SET(AUTOREF_TIMEOUT,60)
TargetMarker(3,1,1)
SET(MSD_MODE,ON)
SET(MSD_THRESHOLD,15)

ACTION MAIN

    SET(COUNTER1,COUNTER_ZERO)
    SET(COUNTER2,COUNTER_ZERO)
    LOAD(ARENAS,"a6.bmp")

    LOGFILE(0,"mwp_ld_3x120s_6w")
    INVOKE(SETUP_HEADERS,1)

    INVOKE(LIGHTS_OFF,1)
    LOAD(OVERLAY_TEXT,"1,1:ACCLIMATION           ")
    WAIT(ACCLIMATION_SEC)
    LOAD(OVERLAY_TEXT,"1,1:AUTOREFERENCING       ")
    AUTOREFERENCE()
    SET(X_DRAWTRACKS,1)
    SET(X_TRACKTIME,15)
    VIDEO(VIDEO_LENGTH,"mwp_ld_3x120s_6w")
    VIDEOSKIP(29)
    INVOKE(LD_CYCLE,NUM_CYCLES)

    VIDEOSTOP
    LOAD(OVERLAY_CLEAR,"")
    INVOKE(LIGHTS_OFF,1)

COMPLETE

ACTION LD_CYCLE

    SET(COUNTER2,COUNTER_INC)
    LOAD(OVERLAY_TEXT,"1,1:PHASE: DARK           ")
    INVOKE(LIGHTS_OFF,1)
    INVOKE(SAMPLE_DARK,PHASE_BINS)

    LOAD(OVERLAY_TEXT,"1,1:PHASE: LIGHT          ")
    INVOKE(LIGHTS_ON,1)
    INVOKE(SAMPLE_LIGHT,PHASE_BINS)

COMPLETE

ACTION SAMPLE_LIGHT

    SET(COUNTER1,COUNTER_INC)
    LOGDATA(DATA_SNAPSHOT,"begin")
    WAIT(TIME_BIN)
    LOGDATA(DATA_SNAPSHOT,"end")
    LOGDATA(DATA_SELECT,"begin")
    LOGDATA(DATA_DELTA,"end")

    LOGCREATE("RUNTIME|APPARATUS_ID|TEMPERATURE1|TEXT:LIGHT|COUNTER2|COUNTER1")
    LOGAPPEND("ARENA_DISTANCES:*")
    LOGAPPEND("ARENA_ACTIVITY:*")
    LOGRUN()

COMPLETE

ACTION SAMPLE_DARK

    SET(COUNTER1,COUNTER_INC)
    LOGDATA(DATA_SNAPSHOT,"begin")
    WAIT(TIME_BIN)
    LOGDATA(DATA_SNAPSHOT,"end")
    LOGDATA(DATA_SELECT,"begin")
    LOGDATA(DATA_DELTA,"end")

    LOGCREATE("RUNTIME|APPARATUS_ID|TEMPERATURE1|TEXT:DARK|COUNTER2|COUNTER1")
    LOGAPPEND("ARENA_DISTANCES:*")
    LOGAPPEND("ARENA_ACTIVITY:*")
    LOGRUN()

COMPLETE

ACTION LIGHTS_ON
    SET(GPO6,1)
    SET(GPO7,1)
    SET(GPO8,1)
COMPLETE

ACTION LIGHTS_OFF
    LIGHTS(ALL,OFF)
    SET(GPO6,0)
    SET(GPO7,0)
    SET(GPO8,0)
COMPLETE

ACTION SETUP_HEADERS

    LOGCREATE("TEXT:RUNTIME|TEXT:UNIT_ID|TEXT:TEMP_C|TEXT:PHASE|TEXT:TRIAL|TEXT:TIME_BIN")
    LOGAPPEND("TEXT:DIST_A1|TEXT:DIST_A2|TEXT:DIST_A3|TEXT:DIST_B1|TEXT:DIST_B2|TEXT:DIST_B3")
    LOGAPPEND("TEXT:MSD_A1|TEXT:MSD_A2|TEXT:MSD_A3|TEXT:MSD_B1|TEXT:MSD_B2|TEXT:MSD_B3")
    LOGRUN()

COMPLETE

ZANTIKS MWP EXPERIMENT BUNDLE
=============================
Researcher       : Not specified
Study Tag        : (none)
Plate Format     : 6-Well (2 x 3, 6 arena(s))
Config Profile   : Default
Plate Geometry   : X_OFFSET=24.76, Y_OFFSET=23.16, COL_STEP=39.12, ROW_STEP=39.12, RADIUS=17.72, INNER_ZONE=12.73 mm
Hardware Target  : MWP Z2
Stimulus Protocol: Light / Dark Cycles
Estimated Runtime: 15m 00s (720 CSV rows)

EXECUTION ORDER IN ZANTIKS CONTROL CONSOLE
------------------------------------------
1. 01_cal.zs
   - Run ONCE with the black 96-well calibration plate. Skip if already calibrated.

2. 02_mwp_6w.zs
   - Run ONCE with an EMPTY 6-Well plate.
   - Generates "a6.bmp" and "dz6.bmp" inside the unit's Assets directory.

3. 03_mwp_6w.zs (Optional)
   - Run with organisms loaded to check TARGET_SIZE and DETECTOR_THRESHOLD on Live View.

4. 04_mwp_6w.zs
   - Main assay script. Load your experimental plate, build, and run.
   - Slot name for Settings -> Slots (1-20): "04_mwp_6w" (link once; rebuilding updates it in place).
   - Exported CSV & video files are tagged automatically via LOGFILE(0,"mwp_ld_3x120s_6w").
   - Download exported .csv data from Control Console -> Data files when complete.

UNIT REMINDERS
--------------
- Do not rename scripts with spaces or parentheses.
- Always click Save before Build in the Zantiks web editor.
- Always shut down via the Control Console webpage before switching off unit power.
