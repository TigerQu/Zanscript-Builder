===== 02_mwp_96sq.zs =====
# Step 2: Plate Arena & Dual-Zone Asset Builder (96-Well Square)
# Config profile: Default MWP
# Place an EMPTY 96-Well Square plate into the MWP stand before running.
# Generates "a96_sq.bmp" (full wells) and "dz96_sq.bmp" (50/50 outer/inner zones) in Assets.

DEFINE SHOWTIME 2
DEFINE DRAWDELAY 0.15

# Plate geometry in mm (8 rows x 12 columns)
DEFINE X_OFFSET 14.15
DEFINE Y_OFFSET 9.65
DEFINE COL_STEP 9
DEFINE ROW_STEP 9
DEFINE LENGTH 7.5
DEFINE LENGTH_DZ 4.5

DEFINE ARENA_ZONE_MODE 200
DEFINE ROW_COUNTER 101
DEFINE ARENA_COUNTER 102
DEFINE COLUMN_COUNTER 103
DEFINE Y_POSITION 104
DEFINE X_POSITION 105

ACTION MAIN

    INVOKE(BUILD_ARENAS,1)
    WAIT(SHOWTIME)

    INVOKE(BUILD_DUAL_ZONES,1)
    WAIT(SHOWTIME)

COMPLETE

ACTION BUILD_ARENAS

    ResetDrawing()
    ShapeType(SQUARE,LENGTH)
    @ARENA_ZONE_MODE = 0
    INVOKE(DRAW_GRID,1)
    SaveDrawing("a96_sq")

COMPLETE

ACTION BUILD_DUAL_ZONES

    ResetDrawing()
    ShapeType(SQUARE,LENGTH)
    @ARENA_ZONE_MODE = 1
    INVOKE(DRAW_GRID,1)

    ShapeType(SQUARE,LENGTH_DZ)
    @ARENA_ZONE_MODE = 2
    INVOKE(DRAW_GRID,1)
    SaveDrawing("dz96_sq")

COMPLETE

ACTION DRAW_GRID

    @ROW_COUNTER = 0
    @Y_POSITION = Y_OFFSET
    @ARENA_COUNTER = 1

    WHILE @ROW_COUNTER < 8
        @COLUMN_COUNTER = 0
        @X_POSITION = X_OFFSET

        WHILE @COLUMN_COUNTER < 12
            IF @ARENA_ZONE_MODE = 0
                SET(DrawArea,@ARENA_COUNTER)
            ELSE
                SET(DrawArea,@ARENA_ZONE_MODE)
            ENDIF

            ShapeDraw(@X_POSITION,@Y_POSITION)
            SaveDrawing("TEMP")
            WAIT(DRAWDELAY)

            @X_POSITION = @X_POSITION + COL_STEP
            @COLUMN_COUNTER = @COLUMN_COUNTER + 1
            @ARENA_COUNTER = @ARENA_COUNTER + 1
        ENDWHILE

        @Y_POSITION = @Y_POSITION + ROW_STEP
        @ROW_COUNTER = @ROW_COUNTER + 1
    ENDWHILE

COMPLETE

    LOGRUN()

    VIDEO(VIDEO_LENGTH,"mwp_track_check_96sq")
    LOAD(OVERLAY_TEXT,"1,1:AUTOREFERENCING       ")
    AUTOREFERENCE()

    INVOKE(SWEEP_SETTINGS,1)

    VIDEOSTOP
    LOAD(OVERLAY_CLEAR,"")
    INVOKE(LIGHTS_OFF,1)

COMPLETE

ACTION SWEEP_SETTINGS

    @SIZE_VAR = MIN_SIZE
    WHILE @SIZE_VAR <= MAX_SIZE
        SET(TARGET_SIZE,@SIZE_VAR)
        TargetMarker(@SIZE_VAR,1,1)
        @THRESH_VAR = MIN_THRESH

        WHILE @THRESH_VAR <= MAX_THRESH
            SET(DETECTOR_THRESHOLD,@THRESH_VAR)
            LOAD(OVERLAY_TEXT,"1,1:SIZE @101  THRESH @102  ")
            INVOKE(SAMPLE_BIN,SAMPLES_PER_SETTING)
            @THRESH_VAR = @THRESH_VAR + 2
        ENDWHILE

        @SIZE_VAR = @SIZE_VAR + 1
    ENDWHILE

COMPLETE

ACTION SAMPLE_BIN

    SET(COUNTER1,COUNTER_INC)
    LOGDATA(DATA_SNAPSHOT,"begin")
    WAIT(TIME_BIN)
    LOGDATA(DATA_SNAPSHOT,"end")
    LOGDATA(DATA_SELECT,"begin")
    LOGDATA(DATA_DELTA,"end")

    LOGCREATE("RUNTIME|APPARATUS_ID|TEMPERATURE1|@101|@102|COUNTER1")
    LOGAPPEND("ARENA_DISTANCES:*")
    LOGRUN()

COMPLETE

ACTION LIGHTS_ON
    SET(X_LED_DRIVE,3,LED_BRIGHTNESS)
    # SET(X_LED_DRIVE,11,LED_BRIGHTNESS)  # Uncomment for underneath LED
    LIGHTS(LIGHT16,ON)
COMPLETE

ACTION LIGHTS_OFF
    SET(X_LED_DRIVE,3,0)
    # SET(X_LED_DRIVE,11,0)
    LIGHTS(LIGHT16,OFF)
COMPLETE


===== 04_mwp_96sq.zs =====
# Step 4: Main Assay Script (96-Well Square | Temperature Step / Ramp)
# Required asset(s) in unit Assets folder: "a96_sq.bmp"

DEFINE ACCLIMATION_SEC 120
DEFINE TIME_BIN 1
DEFINE STEP_BINS 120  # 120s hold per temperature step (3 step(s))
DEFINE VIDEO_LENGTH 999999999
DEFINE X_LED_DRIVE 30039
DEFINE LED_BRIGHTNESS 4095
DEFINE X_DRAWTRACKS 30011
DEFINE X_TRACKTIME 30016

SET(THERMOSTAT,20)
SET(TARGET_SIZE,1)
SET(DETECTOR_THRESHOLD,5)
SET(AUTOREF_MODE,MOVEMENT)
SET(AUTOREF_TIMEOUT,60)
TargetMarker(1,1,1)
SET(MSD_MODE,ON)
SET(MSD_THRESHOLD,5)

ACTION MAIN

    SET(COUNTER1,COUNTER_ZERO)
    SET(COUNTER2,COUNTER_ZERO)
    LOAD(ARENAS,"a96_sq.bmp")

    LOGFILE(0,"mwp_temp_ramp_20to28c_96sq")
    INVOKE(SETUP_HEADERS,1)

    INVOKE(LIGHTS_ON,1)
    LOAD(OVERLAY_TEXT,"1,1:ACCLIM 20C            ")
    WAIT(ACCLIMATION_SEC)
    LOAD(OVERLAY_TEXT,"1,1:AUTOREFERENCING       ")
    AUTOREFERENCE()
    SET(X_DRAWTRACKS,1)
    SET(X_TRACKTIME,15)
    VIDEO(VIDEO_LENGTH,"mwp_temp_ramp_20to28c_96sq")
    VIDEOSKIP(29)
    INVOKE(THERMAL_RAMP,1)

    VIDEOSTOP
    SET(THERMOSTAT,20)  # Restore baseline temperature
    LOAD(OVERLAY_CLEAR,"")
    INVOKE(LIGHTS_OFF,1)

COMPLETE

ACTION THERMAL_RAMP

    SET(COUNTER2,COUNTER_INC)
    SET(THERMOSTAT,20)
    LOAD(OVERLAY_TEXT,"1,1:TEMP TARGET 20C       ")
    INVOKE(SAMPLE_T1,STEP_BINS)

    SET(THERMOSTAT,24)

    SET(THERMOSTAT,28)
    LOAD(OVERLAY_TEXT,"1,1:TEMP TARGET 28C       ")
    INVOKE(SAMPLE_T3,STEP_BINS)

COMPLETE

ZANTIKS MWP EXPERIMENT BUNDLE
=============================
Researcher       : Not specified
Study Tag        : (none)
Plate Format     : 96-Well Square (8 x 12, 96 arena(s))
Config Profile   : Default
Plate Geometry   : X_OFFSET=14.15, Y_OFFSET=9.65, COL_STEP=9, ROW_STEP=9, LENGTH=7.5, INNER_ZONE=4.5 mm
Hardware Target  : MWP Z3
Stimulus Protocol: Temperature Step / Ramp
Estimated Runtime: 9m 00s (360 CSV rows)

EXECUTION ORDER IN ZANTIKS CONTROL CONSOLE
------------------------------------------
1. 01_cal.zs
   - Run ONCE with the black 96-well calibration plate. Skip if already calibrated.

2. 02_mwp_96sq.zs
   - Run ONCE with an EMPTY 96-Well Square plate.
   - Generates "a96_sq.bmp" and "dz96_sq.bmp" inside the unit's Assets directory.

3. 03_mwp_96sq.zs (Optional)
   - Run with organisms loaded to check TARGET_SIZE and DETECTOR_THRESHOLD on Live View.

4. 04_mwp_96sq.zs
   - Main assay script. Load your experimental plate, build, and run.
   - Slot name for Settings -> Slots (1-20): "04_mwp_96sq" (link once; rebuilding updates it in place).
   - Exported CSV & video files are tagged automatically via LOGFILE(0,"mwp_temp_ramp_20to28c_96sq").
   - Download exported .csv data from Control Console -> Data files when complete.

UNIT REMINDERS
--------------
- Do not rename scripts with spaces or parentheses.
- Always click Save before Build in the Zantiks web editor.
- Always shut down via the Control Console webpage before switching off unit power.
