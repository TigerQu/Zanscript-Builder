===== 02_assets_48well.zs =====
# Step 2: Plate Arena & Dual-Zone Asset Builder (48-Well)
# Config profile: Default MWP
# Place an EMPTY 48-Well plate into the MWP stand before running.
# Generates "a48.bmp" (full wells) and "dz48.bmp" (50/50 outer/inner zones) in Assets.

DEFINE SHOWTIME 2
DEFINE DRAWDELAY 0.2

# Plate geometry in mm (6 rows x 8 columns)
DEFINE X_OFFSET 18.4
DEFINE Y_OFFSET 10.28
DEFINE COL_STEP 12.92
DEFINE ROW_STEP 12.92
DEFINE RADIUS 5.19
DEFINE RADIUS_DZ 3.45

DEFINE ARENA_ZONE_MODE 200
DEFINE ROW_COUNTER 101
DEFINE ARENA_COUNTER 102
DEFINE COLUMN_COUNTER 103
DEFINE Y_POSITION 104
DEFINE X_POSITION 105

ACTION MAIN

    INVOKE(BUILD_ARENAS,1)
    WAIT(SHOWTIME)

    INVOKE(BUILD_DUAL_ZONES,1)
    WAIT(SHOWTIME)

COMPLETE

ACTION BUILD_ARENAS

    ResetDrawing()
    ShapeType(DISC,RADIUS)
    @ARENA_ZONE_MODE = 0
    INVOKE(DRAW_GRID,1)
    SaveDrawing("a48")

COMPLETE

ACTION BUILD_DUAL_ZONES

    ResetDrawing()
    ShapeType(DISC,RADIUS)
    @ARENA_ZONE_MODE = 1
    INVOKE(DRAW_GRID,1)

    ShapeType(DISC,RADIUS_DZ)
    @ARENA_ZONE_MODE = 2
    INVOKE(DRAW_GRID,1)
    SaveDrawing("dz48")

COMPLETE

ACTION DRAW_GRID

    @ROW_COUNTER = 0
    @Y_POSITION = Y_OFFSET
    @ARENA_COUNTER = 1

    WHILE @ROW_COUNTER < 6
        @COLUMN_COUNTER = 0
        @X_POSITION = X_OFFSET

        WHILE @COLUMN_COUNTER < 8
            IF @ARENA_ZONE_MODE = 0
                SET(DrawArea,@ARENA_COUNTER)
            ELSE
                SET(DrawArea,@ARENA_ZONE_MODE)
            ENDIF

            ShapeDraw(@X_POSITION,@Y_POSITION)
            SaveDrawing("TEMP")
            WAIT(DRAWDELAY)

            @X_POSITION = @X_POSITION + COL_STEP
            @COLUMN_COUNTER = @COLUMN_COUNTER + 1
            @ARENA_COUNTER = @ARENA_COUNTER + 1
        ENDWHILE

        @Y_POSITION = @Y_POSITION + ROW_STEP
        @ROW_COUNTER = @ROW_COUNTER + 1
    ENDWHILE

COMPLETE


===== 03_track_check_48well.zs =====
# Step 3: Organism Tracking Check (48-Well)
# Optional 1-2 minute pre-check with animals loaded in the plate.
# Sweeps TARGET_SIZE (3 to 6) and DETECTOR_THRESHOLD (2 to 10)
# so you can confirm clean target lock on video before running Step 4.

DEFINE TIME_BIN 1
DEFINE SAMPLES_PER_SETTING 2
DEFINE MIN_SIZE 3
DEFINE MAX_SIZE 6
DEFINE MIN_THRESH 2
DEFINE MAX_THRESH 10
DEFINE AUTOREF_TIME 60
DEFINE VIDEO_LENGTH 999999
DEFINE X_LED_DRIVE 30039
DEFINE LED_BRIGHTNESS 4095
SET(THERMOSTAT,22)

DEFINE SIZE_VAR 101
DEFINE THRESH_VAR 102

SET(AUTOREF_MODE,MOVEMENT)
SET(AUTOREF_TIMEOUT,AUTOREF_TIME)

ACTION MAIN

    SET(COUNTER1,COUNTER_ZERO)
    INVOKE(LIGHTS_ON,1)
    LOAD(ARENAS,"a48.bmp")

    LOGCREATE("TEXT:RUNTIME|TEXT:UNIT_ID|TEXT:TEMP_C|TEXT:TARGET_SIZE|TEXT:DET_THRESHOLD|TEXT:TIME_BIN")
    LOGAPPEND("TEXT:DIST_A1|TEXT:DIST_A2|TEXT:DIST_A3|TEXT:DIST_A4|TEXT:DIST_A5|TEXT:DIST_A6")
    LOGAPPEND("TEXT:DIST_A7|TEXT:DIST_A8|TEXT:DIST_B1|TEXT:DIST_B2|TEXT:DIST_B3|TEXT:DIST_B4")
    LOGAPPEND("TEXT:DIST_B5|TEXT:DIST_B6|TEXT:DIST_B7|TEXT:DIST_B8|TEXT:DIST_C1|TEXT:DIST_C2")
    LOGAPPEND("TEXT:DIST_C3|TEXT:DIST_C4|TEXT:DIST_C5|TEXT:DIST_C6|TEXT:DIST_C7|TEXT:DIST_C8")
    LOGAPPEND("TEXT:DIST_D1|TEXT:DIST_D2|TEXT:DIST_D3|TEXT:DIST_D4|TEXT:DIST_D5|TEXT:DIST_D6")
    LOGAPPEND("TEXT:DIST_D7|TEXT:DIST_D8|TEXT:DIST_E1|TEXT:DIST_E2|TEXT:DIST_E3|TEXT:DIST_E4")
    LOGAPPEND("TEXT:DIST_E5|TEXT:DIST_E6|TEXT:DIST_E7|TEXT:DIST_E8|TEXT:DIST_F1|TEXT:DIST_F2")
    LOGAPPEND("TEXT:DIST_F3|TEXT:DIST_F4|TEXT:DIST_F5|TEXT:DIST_F6|TEXT:DIST_F7|TEXT:DIST_F8")
    LOGRUN()

    VIDEO(VIDEO_LENGTH,"mwp_track_check_48well")
    LOAD(OVERLAY_TEXT,"1,1:AUTOREFERENCING       ")
    AUTOREFERENCE()

    INVOKE(SWEEP_SETTINGS,1)

    VIDEOSTOP
    LOAD(OVERLAY_CLEAR,"")
    INVOKE(LIGHTS_OFF,1)

COMPLETE

ACTION SWEEP_SETTINGS

    @SIZE_VAR = MIN_SIZE
    WHILE @SIZE_VAR <= MAX_SIZE
        SET(TARGET_SIZE,@SIZE_VAR)
        TargetMarker(@SIZE_VAR,1,1)
        @THRESH_VAR = MIN_THRESH

        WHILE @THRESH_VAR <= MAX_THRESH
            SET(DETECTOR_THRESHOLD,@THRESH_VAR)
            LOAD(OVERLAY_TEXT,"1,1:SIZE @101  THRESH @102  ")
            INVOKE(SAMPLE_BIN,SAMPLES_PER_SETTING)
            @THRESH_VAR = @THRESH_VAR + 2
        ENDWHILE

        @SIZE_VAR = @SIZE_VAR + 1
    ENDWHILE

COMPLETE

ACTION SAMPLE_BIN

    SET(COUNTER1,COUNTER_INC)
    LOGDATA(DATA_SNAPSHOT,"begin")
    WAIT(TIME_BIN)
    LOGDATA(DATA_SNAPSHOT,"end")
    LOGDATA(DATA_SELECT,"begin")
    LOGDATA(DATA_DELTA,"end")

    LOGCREATE("RUNTIME|APPARATUS_ID|TEMPERATURE1|@101|@102|COUNTER1")
    LOGAPPEND("ARENA_DISTANCES:*")
    LOGRUN()

COMPLETE

ACTION LIGHTS_ON
    SET(X_LED_DRIVE,4,LED_BRIGHTNESS)
    # SET(X_LED_DRIVE,12,LED_BRIGHTNESS)  # Uncomment for underneath LED
COMPLETE

ACTION LIGHTS_OFF
    SET(X_LED_DRIVE,4,0)
    # SET(X_LED_DRIVE,12,0)
COMPLETE


===== 04_mwp_vibration_startle_48well.zs =====
# Step 4: Main Assay Script (48-Well | Vibration Startle)
# Required asset(s) in unit Assets folder: "a48.bmp"

DEFINE ACCLIMATION_SEC 120
DEFINE TIME_BIN 1
DEFINE BASELINE_BINS 60
DEFINE NUM_PULSES 4
DEFINE ITI_BINS 30
DEFINE VIDEO_LENGTH 999999999
DEFINE X_LED_DRIVE 30039
DEFINE LED_BRIGHTNESS 4095
DEFINE X_DRAWTRACKS 30011
DEFINE X_TRACKTIME 30016

SET(THERMOSTAT,22)
SET(TARGET_SIZE,4)
SET(DETECTOR_THRESHOLD,8)
SET(AUTOREF_MODE,MOVEMENT)
SET(AUTOREF_TIMEOUT,60)
TargetMarker(2.4,1,1)
SET(MSD_MODE,ON)
SET(MSD_THRESHOLD,6)

ACTION MAIN

    SET(COUNTER1,COUNTER_ZERO)
    SET(COUNTER2,COUNTER_ZERO)
    LOAD(ARENAS,"a48.bmp")

    INVOKE(SETUP_HEADERS,1)

    INVOKE(LIGHTS_ON,1)
    LOAD(OVERLAY_TEXT,"1,1:ACCLIMATION           ")
    WAIT(ACCLIMATION_SEC)
    LOAD(OVERLAY_TEXT,"1,1:AUTOREFERENCING       ")
    AUTOREFERENCE()
    SET(X_DRAWTRACKS,1)
    SET(X_TRACKTIME,15)
    VIDEO(VIDEO_LENGTH,"mwp_vibration_startle_48well")
    VIDEOSKIP(29)
    LOAD(OVERLAY_TEXT,"1,1:BASELINE              ")
    INVOKE(SAMPLE_BASELINE,BASELINE_BINS)
    INVOKE(VIBRATION_TRIAL,NUM_PULSES)

    VIDEOSTOP
    LOAD(OVERLAY_CLEAR,"")
    INVOKE(LIGHTS_OFF,1)

COMPLETE

ACTION VIBRATION_TRIAL

    SET(COUNTER2,COUNTER_INC)
    INVOKE(VIBRATION_PULSE,1)
    LOAD(OVERLAY_TEXT,"1,1:POST-VIB ITI          ")
    INVOKE(SAMPLE_ITI,ITI_BINS)

COMPLETE

ACTION VIBRATION_PULSE

    LOGAPPEND("ARENA_ACTIVITY:*")
