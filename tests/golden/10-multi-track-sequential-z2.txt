===== 02_mwp_24w.zs =====
# Step 2: Plate Arena & Dual-Zone Asset Builder (24-Well)
# Config profile: Default MWP
# Place an EMPTY 24-Well plate into the MWP stand before running.
# Generates "a24.bmp" (full wells) and "dz24.bmp" (50/50 outer/inner zones) in Assets.

DEFINE SHOWTIME 2
DEFINE DRAWDELAY 0.25

# Plate geometry in mm (4 rows x 6 columns)
DEFINE X_OFFSET 15.2
DEFINE Y_OFFSET 13.5
DEFINE COL_STEP 19.47
DEFINE ROW_STEP 19.16
DEFINE RADIUS 8.13
DEFINE RADIUS_DZ 5.6

DEFINE ARENA_ZONE_MODE 200
DEFINE ROW_COUNTER 101
DEFINE ARENA_COUNTER 102
DEFINE COLUMN_COUNTER 103
DEFINE Y_POSITION 104
DEFINE X_POSITION 105

ACTION MAIN

    INVOKE(BUILD_ARENAS,1)
    WAIT(SHOWTIME)

    INVOKE(BUILD_DUAL_ZONES,1)
    WAIT(SHOWTIME)

COMPLETE

ACTION BUILD_ARENAS

    ResetDrawing()
    ShapeType(DISC,RADIUS)
    @ARENA_ZONE_MODE = 0
    INVOKE(DRAW_GRID,1)
    SaveDrawing("a24")

COMPLETE

ACTION BUILD_DUAL_ZONES

    ResetDrawing()
    ShapeType(DISC,RADIUS)
    @ARENA_ZONE_MODE = 1
    INVOKE(DRAW_GRID,1)

    ShapeType(DISC,RADIUS_DZ)
    @ARENA_ZONE_MODE = 2
    INVOKE(DRAW_GRID,1)
    SaveDrawing("dz24")

COMPLETE

ACTION DRAW_GRID

    @ROW_COUNTER = 0
    @Y_POSITION = Y_OFFSET
    @ARENA_COUNTER = 1

    WHILE @ROW_COUNTER < 4
        @COLUMN_COUNTER = 0
        @X_POSITION = X_OFFSET

        WHILE @COLUMN_COUNTER < 6
            IF @ARENA_ZONE_MODE = 0
                SET(DrawArea,@ARENA_COUNTER)
            ELSE
                SET(DrawArea,@ARENA_ZONE_MODE)
            ENDIF

            ShapeDraw(@X_POSITION,@Y_POSITION)
            SaveDrawing("TEMP")
            WAIT(DRAWDELAY)

            @X_POSITION = @X_POSITION + COL_STEP
            @COLUMN_COUNTER = @COLUMN_COUNTER + 1
            @ARENA_COUNTER = @ARENA_COUNTER + 1
        ENDWHILE

        @Y_POSITION = @Y_POSITION + ROW_STEP
        @ROW_COUNTER = @ROW_COUNTER + 1
    ENDWHILE

COMPLETE


===== 03_mwp_24w.zs =====
# Step 3: Organism Tracking Check (24-Well)
# Optional 1-2 minute pre-check with animals loaded in the plate.
# Sweeps TARGET_SIZE (3 to 6) and DETECTOR_THRESHOLD (2 to 10)
# so you can confirm clean target lock on video before running Step 4.

DEFINE TIME_BIN 1
DEFINE SAMPLES_PER_SETTING 2
DEFINE MIN_SIZE 3
DEFINE MAX_SIZE 6
DEFINE MIN_THRESH 2
DEFINE MAX_THRESH 10
DEFINE AUTOREF_TIME 60
DEFINE VIDEO_LENGTH 999999
SET(THERMOSTAT,22)

DEFINE SIZE_VAR 101
DEFINE THRESH_VAR 102

SET(AUTOREF_MODE,MOVEMENT)
SET(AUTOREF_TIMEOUT,AUTOREF_TIME)

ACTION MAIN

    SET(COUNTER1,COUNTER_ZERO)
    INVOKE(LIGHTS_OFF,1)
    LOAD(ARENAS,"a24.bmp")

    LOGFILE(0,"mwp_track_check_24w")
    LOGCREATE("TEXT:RUNTIME|TEXT:UNIT_ID|TEXT:TEMP_C|TEXT:TARGET_SIZE|TEXT:DET_THRESHOLD|TEXT:TIME_BIN")
    LOGAPPEND("TEXT:DIST_A1|TEXT:DIST_A2|TEXT:DIST_A3|TEXT:DIST_A4|TEXT:DIST_A5|TEXT:DIST_A6")
    LOGAPPEND("TEXT:DIST_B1|TEXT:DIST_B2|TEXT:DIST_B3|TEXT:DIST_B4|TEXT:DIST_B5|TEXT:DIST_B6")
    LOGAPPEND("TEXT:DIST_C1|TEXT:DIST_C2|TEXT:DIST_C3|TEXT:DIST_C4|TEXT:DIST_C5|TEXT:DIST_C6")
    LOGAPPEND("TEXT:DIST_D1|TEXT:DIST_D2|TEXT:DIST_D3|TEXT:DIST_D4|TEXT:DIST_D5|TEXT:DIST_D6")
    LOGRUN()

    VIDEO(VIDEO_LENGTH,"mwp_track_check_24w")
    LOAD(OVERLAY_TEXT,"1,1:AUTOREFERENCING       ")
    AUTOREFERENCE()

    INVOKE(SWEEP_SETTINGS,1)

    VIDEOSTOP
    LOAD(OVERLAY_CLEAR,"")
    INVOKE(LIGHTS_OFF,1)

COMPLETE

ACTION SWEEP_SETTINGS

    @SIZE_VAR = MIN_SIZE
    WHILE @SIZE_VAR <= MAX_SIZE
        SET(TARGET_SIZE,@SIZE_VAR)
        TargetMarker(@SIZE_VAR,1,1)
        @THRESH_VAR = MIN_THRESH

        WHILE @THRESH_VAR <= MAX_THRESH
            SET(DETECTOR_THRESHOLD,@THRESH_VAR)
            LOAD(OVERLAY_TEXT,"1,1:SIZE @101  THRESH @102  ")
            INVOKE(SAMPLE_BIN,SAMPLES_PER_SETTING)
            @THRESH_VAR = @THRESH_VAR + 2
        ENDWHILE

        @SIZE_VAR = @SIZE_VAR + 1
    ENDWHILE

COMPLETE

ACTION SAMPLE_BIN

    SET(COUNTER1,COUNTER_INC)
    LOGDATA(DATA_SNAPSHOT,"begin")
    WAIT(TIME_BIN)
    LOGDATA(DATA_SNAPSHOT,"end")
    LOGDATA(DATA_SELECT,"begin")
    LOGDATA(DATA_DELTA,"end")

    LOGCREATE("RUNTIME|APPARATUS_ID|TEMPERATURE1|@101|@102|COUNTER1")
    LOGAPPEND("ARENA_DISTANCES:*")
    LOGRUN()

COMPLETE

ACTION LIGHTS_ON
    SET(GPO6,1)
    SET(GPO7,1)
    SET(GPO8,1)
COMPLETE

ACTION LIGHTS_OFF
    LIGHTS(ALL,OFF)
    SET(GPO6,0)
    SET(GPO7,0)
    SET(GPO8,0)
COMPLETE

# Step 4: Main Assay Script (24-Well | Multi-Track Timeline)
# Required asset(s) in unit Assets folder: "a24.bmp"

DEFINE ACCLIMATION_SEC 120
DEFINE TIME_BIN 1
DEFINE NUM_CYCLES 1
DEFINE S1_BINS 20  # 0s-20s (S1_DARK)
DEFINE S2_BINS 30  # 20s-50s (S2_BLUE)
DEFINE S3_BINS 25  # 50s-75s (S3_DARK)
DEFINE S4_BINS 30  # 75s-105s (S4_GREEN)
DEFINE S5_BINS 25  # 105s-130s (S5_DARK)
DEFINE S6_BINS 30  # 130s-160s (S6_RED)
DEFINE S7_BINS 20  # 160s-180s (S7_DARK)
DEFINE VIDEO_LENGTH 999999999
DEFINE X_DRAWTRACKS 30011
DEFINE X_TRACKTIME 30016

SET(THERMOSTAT,22)
SET(TARGET_SIZE,4)
SET(DETECTOR_THRESHOLD,8)
SET(AUTOREF_MODE,MOVEMENT)
SET(AUTOREF_TIMEOUT,60)
TargetMarker(2.4,1,1)
SET(MSD_MODE,ON)
SET(MSD_THRESHOLD,6)

ACTION MAIN

    SET(COUNTER1,COUNTER_ZERO)
    SET(COUNTER2,COUNTER_ZERO)
    LOAD(ARENAS,"a24.bmp")

    LOGFILE(0,"mwp_multi_track_1x180s_24w")
    INVOKE(SETUP_HEADERS,1)

    LIGHTS(ALL,OFF)
    SET(GPO6,0)
    SET(GPO7,0)
    SET(GPO8,0)
    LOAD(OVERLAY_TEXT,"1,1:ACCLIMATION           ")
    WAIT(ACCLIMATION_SEC)
    LOAD(OVERLAY_TEXT,"1,1:AUTOREFERENCING       ")
    AUTOREFERENCE()
    SET(X_DRAWTRACKS,1)
    SET(X_TRACKTIME,15)
    VIDEO(VIDEO_LENGTH,"mwp_multi_track_1x180s_24w")
    VIDEOSKIP(29)
    INVOKE(TIMELINE_CYCLE,NUM_CYCLES)

    VIDEOSTOP
    LOAD(OVERLAY_CLEAR,"")
    INVOKE(LIGHTS_OFF,1)

COMPLETE

ACTION TIMELINE_CYCLE

===== README_STEPS.txt =====
ZANTIKS MWP EXPERIMENT BUNDLE
=============================
Researcher       : Not specified
Study Tag        : (none)
Plate Format     : 24-Well (4 x 6, 24 arena(s))
Config Profile   : Default
Plate Geometry   : X_OFFSET=15.2, Y_OFFSET=13.5, COL_STEP=19.47, ROW_STEP=19.16, RADIUS=8.13, INNER_ZONE=5.6 mm
Hardware Target  : MWP Z2
Stimulus Protocol: Multi-Track Timeline
Estimated Runtime: 6m 00s (180 CSV rows)

EXECUTION ORDER IN ZANTIKS CONTROL CONSOLE
------------------------------------------
1. 01_cal.zs
   - Run ONCE with the black 96-well calibration plate. Skip if already calibrated.

2. 02_mwp_24w.zs
   - Run ONCE with an EMPTY 24-Well plate.
   - Generates "a24.bmp" and "dz24.bmp" inside the unit's Assets directory.

3. 03_mwp_24w.zs (Optional)
   - Run with organisms loaded to check TARGET_SIZE and DETECTOR_THRESHOLD on Live View.

4. 04_mwp_24w.zs
   - Main assay script. Load your experimental plate, build, and run.
   - Slot name for Settings -> Slots (1-20): "04_mwp_24w" (link once; rebuilding updates it in place).
   - Exported CSV & video files are tagged automatically via LOGFILE(0,"mwp_multi_track_1x180s_24w").
   - Download exported .csv data from Control Console -> Data files when complete.

UNIT REMINDERS
--------------
- Do not rename scripts with spaces or parentheses.
- Always click Save before Build in the Zantiks web editor.
- Always shut down via the Control Console webpage before switching off unit power.
